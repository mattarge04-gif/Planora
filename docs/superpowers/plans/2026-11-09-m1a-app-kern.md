# M1a App-Kern Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Lauffaehiger Server-Kern einer Planora-Instanz: Migrator, DB-Rollen, Audit mit Pseudonym, Modulmanifest mit Guard und Schaltern, lokaler Login (Better Auth) mit Sitzungsregeln und Rechtematrix, Dateiablage – alles lokal mit Tests nachgewiesen.

**Architecture:** Ein Node-Paket `app/` (TypeScript, Fastify 5). SQL-Migrationen als Dateien, ausgefuehrt von einem eigenen Migrator mit Advisory-Lock und Journal; Drizzle nur als Query-Builder fuer Better Auth. Tests mit Vitest gegen echtes PostgreSQL 18 aus `embedded-postgres` (kein Docker noetig). Betrieb (Compose, Backup, Update) ist Plan M1b.

**Tech Stack:** Node 24, TypeScript, Fastify 5.12.8, better-auth 1.7.18, drizzle-orm 0.45.4, pg 8.23.1, @node-rs/argon2 2.2.2, embedded-postgres 18.6.0-beta.19 (Versionen aus Spike S13); Vitest/TypeScript/tsx: exakte Version beim Installieren pinnen (`npm i -E`).

**Spec:** `docs/superpowers/specs/2026-11-09-m1-server-paket-design.md` (V4; §15 > §14 > §4–§8). Spike-Muster: `spikes/s13-m1-kern/t1_auth.mjs`, `db.mjs`.

`app/src/db/migrator.ts`:
```ts
import pg from "pg";
import { createHash } from "node:crypto";
import { readdirSync, readFileSync } from "node:fs";
import { join } from "node:path";
import { fileURLToPath } from "node:url";

const STANDARD = fileURLToPath(new URL("./migrationen", import.meta.url));
const LOCK_ID = 7_310_001; // fester Advisory-Lock fuer Migrationen

export async function migriere(cfg: pg.ClientConfig, verzeichnis = STANDARD) {
  const c = new pg.Client(cfg); await c.connect();
  const angewendet: string[] = [];
  try {
    await c.query("select pg_advisory_lock($1)", [LOCK_ID]);
    await c.query(`create table if not exists migration_journal(name text primary key, pruefsumme text not null, angewendet_am timestamptz not null default now())`);
    const dateien = readdirSync(verzeichnis).filter((f) => f.endsWith(".sql")).sort();
    const bekannt = new Map((await c.query("select name, pruefsumme from migration_journal")).rows.map((r) => [r.name, r.pruefsumme]));
    for (const f of dateien) {
      const sql = readFileSync(join(verzeichnis, f), "utf8");
      const summe = createHash("sha256").update(sql).digest("hex");
      if (bekannt.has(f)) { if (bekannt.get(f) !== summe) throw new Error(`Pruefsumme geaendert: ${f}`); continue; }
      await c.query("begin");
      try { await c.query(sql); await c.query("insert into migration_journal(name, pruefsumme) values ($1,$2)", [f, summe]); await c.query("commit"); }
      catch (e) { await c.query("rollback"); throw e; }
      angewendet.push(f);
    }
  } finally { await c.query("select pg_advisory_unlock($1)", [LOCK_ID]).catch(() => {}); await c.end(); }
  return { angewendet };
}

if (process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1]) {
  migriere({ connectionString: process.env.PLANORA_MIGRATOR_DB }).then((r) => console.log(r), (e) => { console.error(e.message); process.exit(1); });
}
```

import { auditEreignis } from "../audit/audit.js";

export async function aktiveOptionen(c: pg.ClientBase) {
  return new Set<ModulId>((await c.query("select modul_id from modul_schalter where aktiv")).rows.map((r) => r.modul_id));
}
export async function modulAktiv(c: pg.ClientBase, id: ModulId, stufe: Stufe) { return istAktiv(id, await aktiveOptionen(c), stufe); }

export async function schalteEin(pool: pg.Pool, id: ModulId, stufe: Stufe, auditId: string) {
  const c = await pool.connect();
  try {
    await c.query("begin"); await c.query("lock table modul_schalter in share row exclusive mode");
    const r = pruefeEinschalten(await aktiveOptionen(c), id, stufe); if (!r.ok) throw new Error(r.grund);
    await c.query(`insert into modul_schalter(modul_id, aktiv, geaendert_von, manifest_version) values ($1,true,$2,$3)
      on conflict (modul_id) do update set aktiv=true, geaendert_am=now(), geaendert_von=$2, manifest_version=$3`, [id, auditId, MANIFEST_VERSION]);
    await auditEreignis(c, auditId, "modul.ein", { id }); await c.query("commit");
  } catch (e) { await c.query("rollback"); throw e; } finally { c.release(); }
}
export async function schalteAus(pool: pg.Pool, id: ModulId, auditId: string) {
  const c = await pool.connect();
  try {
    await c.query("begin"); await c.query("lock table modul_schalter in share row exclusive mode");
    const raus = kaskadeAbschalten(await aktiveOptionen(c), id);
    for (const m of raus) { await c.query("update modul_schalter set aktiv=false, geaendert_am=now(), geaendert_von=$2 where modul_id=$1", [m, auditId]); await auditEreignis(c, auditId, "modul.aus", { id: m, ausgeloest_durch: id }); }

  } catch (e) { await c.query("rollback"); throw e; } finally { c.release(); }
}

  });
}

import { createHash, randomBytes } from "node:crypto";
import { createReadStream, createWriteStream, promises as fs } from "node:fs";
import { dirname, join, resolve, sep } from "node:path";
import { pipeline } from "node:stream/promises";
import { Transform, type Readable } from "node:stream";

export type Ablage = { schreibeNeu(d: Readable): Promise<{ schluessel: string; sha256: string; groesse: number }>; lese(s: string, range?: { start: number; ende: number }): Readable; existiert(s: string): Promise<boolean>; entferne(s: string): Promise<void> };
const MUSTER = /^sha256\/[0-9a-f]{2}\/[0-9a-f]{2}\/[0-9a-f]{64}$/;

export function dateiAblage(wurzel: string, opts: { maxBelegung?: number; belegung?: () => Promise<number> } = {}): Ablage {
  const basis = resolve(wurzel); const max = opts.maxBelegung ?? 0.9;
  const belegung = opts.belegung ?? (async () => { const s = await fs.statfs(basis); return 1 - s.bavail / s.blocks; });
  const pfad = (s: string) => { if (!MUSTER.test(s)) throw new Error("ungueltiger Schluessel"); const p = resolve(basis, s); if (!p.startsWith(basis + sep)) throw new Error("ungueltiger Schluessel"); return p; };
  return {
    async schreibeNeu(d) {
      await fs.mkdir(join(basis, "tmp"), { recursive: true });
      if ((await belegung()) >= max) throw new Error("Speicher voll");
      const tmp = join(basis, "tmp", randomBytes(12).toString("hex")); const h = createHash("sha256"); let groesse = 0;
      await pipeline(d, new Transform({ transform(ch, _e, cb) { h.update(ch); groesse += ch.length; cb(null, ch); } }), createWriteStream(tmp, { flags: "wx" }));
      const sha256 = h.digest("hex"); const schluessel = `sha256/${sha256.slice(0, 2)}/${sha256.slice(2, 4)}/${sha256}`; const ziel = pfad(schluessel);
      await fs.mkdir(dirname(ziel), { recursive: true });
      try { await fs.access(ziel); await fs.rm(tmp); } catch { await fs.rename(tmp, ziel); await fs.chmod(ziel, 0o440); }
      return { schluessel, sha256, groesse };
    },
    lese(s, range) { return createReadStream(pfad(s), range ? { start: range.start, end: range.ende } : undefined); },
    async existiert(s) { try { await fs.access(pfad(s)); return true; } catch { return false; } },
    async entferne(s) { const p = pfad(s); await fs.chmod(p, 0o640).catch(() => {}); await fs.rm(p, { force: true }); },
  };
}

import { registriereModulGuard, schalteEin, schalteAus, aktiveOptionen } from "./module/schalter.js";
import { MANIFEST, geliefert, type ModulId, type Stufe } from "./module/manifest.js";
import { pruefe } from "./rechte/rechte.js";

export async function baueServer(o: { pool: pg.Pool; stufe: Stufe; baseURL: string; secret: string; ablageWurzel: string }) {
  const app = Fastify({ logger: false });
  const auth = baueAuth(o.pool, { baseURL: o.baseURL, secret: o.secret });
  registriereAuth(app, auth, o.baseURL);
  registriereModulGuard(app, o.pool, o.stufe);
  registriereSitzungsregeln(app, auth, o.pool, [o.baseURL]);
  const darf = async (benutzerId: string, recht: Parameters<typeof pruefe>[2]) => { const c = await o.pool.connect(); try { return await pruefe(c, benutzerId, recht); } finally { c.release(); } };
  app.get("/api/ich", async (req) => ({ id: req.kontext!.benutzerId }));
  app.get("/api/admin/module", async (req, reply) => {
    if (!(await darf(req.kontext!.benutzerId, "module.schalten"))) return reply.code(403).send();
    const c = await o.pool.connect(); try { const aktiv = await aktiveOptionen(c); return geliefert(o.stufe).map((m) => ({ id: m.id, art: m.art, aktiv: m.art === "kern" || aktiv.has(m.id) })); } finally { c.release(); }
  });
  app.post<{ Params: { id: ModulId; aktion: "ein"|"aus" } }>("/api/admin/module/:id/:aktion", async (req, reply) => {
    if (!(await darf(req.kontext!.benutzerId, "module.schalten"))) return reply.code(403).send();
    if (!MANIFEST.some((m) => m.id === req.params.id)) return reply.code(404).send();
    try { return req.params.aktion === "ein" ? (await schalteEin(o.pool, req.params.id, o.stufe, req.kontext!.auditId), { ok: true }) : { aus: await schalteAus(o.pool, req.params.id, req.kontext!.auditId) }; }
    catch (e) { return reply.code(409).send({ fehler: (e as Error).message }); }
  });
  for (const m of MANIFEST) for (const r of m.routen) if (!r.startsWith("/api/auth/") && !r.startsWith("/api/admin/")) app.all(`${r}*`, async (_req, reply) => reply.code(501).send({ fehler: "noch nicht umgesetzt" }));
  return app;
}
